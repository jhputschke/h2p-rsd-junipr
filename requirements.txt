# Core runtime dependencies for h2p-rsd-junipr (training + evaluation + the
# RNTuple data path). Pinned to the tested environment for reproducibility.
#
#   pip install -r requirements.txt
#
# The package itself is installed with `pip install -e .` (see pyproject.toml);
# this file mirrors the [project].dependencies for users who want a plain
# requirements-based environment. Optional extras live in requirements-dev.txt.

# torch: a >= floor rather than an exact pin — install the right wheel for your
# platform/accelerator from https://pytorch.org (tested with 2.11.0.dev; >=2.1
# is the supported floor). On Apple Silicon the default wheel includes MPS.
torch>=2.10

numpy>=2.0,<3        # tested: 2.3.5
omegaconf>=2.3,<3     # tested: 2.3.1  (structured-config validation, no Hydra)
uproot>=5.0,<6        # tested: 5.6.9  (reads the ROOT RNTuple written by cpp/)
awkward>=2.6,<3       # tested: 2.8.11 (jagged Lund columns: arrays(library="ak"))

# point_estimator=map path (and per_jet_nll parity) stays dependency-free. Mirrors the
# `[mbr]` extra in pyproject.toml.
pot>=0.9             # tested: 0.9.7  (ot.emd2 augmented-cost EMD)
