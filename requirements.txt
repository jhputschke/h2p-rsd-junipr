# Core runtime dependencies for h2p-rsd-junipr (training + evaluation + the
# RNTuple data path). Pinned to the tested environment for reproducibility.
#
#   pip install -r requirements.txt
#
# The package itself is installed with `pip install -e .` (see pyproject.toml);
# this file covers the same [project].dependencies for users who want a plain
# requirements-based environment. Optional extras live in requirements-dev.txt.
#
# Floors here track the *tested* environment and are deliberately tighter than
# pyproject's [project].dependencies, which state the lowest versions the code is
# expected to work on (torch>=2.1, numpy>=1.24 — what docker/Dockerfile.train's
# torch-2.4 base image resolves to). Install from pyproject if you need those.

# torch: a >= floor rather than an exact pin — install the right wheel for your
# platform/accelerator from https://pytorch.org (tested with 2.11.0.dev).
# On Apple Silicon the default wheel includes MPS.
torch>=2.10

numpy>=2.0,<3        # tested: 2.3.5  (nothing here needs the 2.x API; see above)
omegaconf>=2.3,<3     # tested: 2.3.1  (structured-config validation, no Hydra)
uproot>=5.0,<6        # tested: 5.6.9  (reads the ROOT RNTuple written by cpp/)
awkward>=2.6,<3       # tested: 2.8.11 (jagged Lund columns: arrays(library="ak"))

# --- MBR point estimator (decode.point_estimator=mbr) ---
# `pot` is the DEFAULT optimal-transport backend (decode.mbr_backend=pot). It is
# lazy-imported, so it is only needed when MBR is actually selected — the default
# point_estimator=map path (and per_jet_nll parity) stays dependency-free. Mirrors the
# `[mbr]` extra in pyproject.toml.
pot>=0.9             # tested: 0.9.7  (ot.emd2 augmented-cost EMD)
# `energyflow` is the OPTIONAL reference EMD backend (decode.mbr_backend=energyflow),
# heavier (pulls the Wasserstein/POT stack). Uncomment, or `pip install -e ".[energyflow]"`,
# to enable it; it too is lazy-imported and only needed when that backend is chosen.
# energyflow>=1.3    # reference KMT EMD (arXiv:1902.02346); needs a working `wasserstein` build
